public final class MatchModeInstance implements AutoCloseable {

    private final MatchModeInstanceId id;
    private final Session session;
    private final MatchMode mode;                   // 只读:模板信息
    private final MapDefinition map;
    private final MatchModeConfig cfg;
    private final Handoff handoff;
    private final MatchModeInstanceClock clock;
    private final ComponentRegistry components;   // 有序:id → 组件实例(声明序)
    private final MatchModeInstanceEventBus bus;
    private final Map<UUID, Participant> participants = new LinkedHashMap<>();
    private final MatchModeInstanceServices svc;           // 组件共享的服务集(self/components/modeInfo)
    private volatile boolean closed;
    private volatile boolean ending;              // svc().self().end() 置位

    /** SessionService 切换阶段 B4 调用(经 MatchMode#createInstance 进来,依赖已校验)。 */
    void awakeAll() {
        components.inDeclarationOrder().forEach(c -> c.awake());        // 只解析依赖,玩家不可见
    }
    void startAll() {
        participants.putAll(session.roster().toParticipants());          // 名册原地位移
        components.inDeclarationOrder().forEach(c -> c.start());         // 传送/发装备/登记显示
        bus.publish(new MatchModeInstanceStarted(clock.tick()));
    }

    /** 由全局 EngineTicker 每 tick 调用。★ 主线程。只派发,不解释。 */
    public void tick() {
        if (closed) return;
        long now = clock.tick();
        try {
            components.forEachTicking(c -> c.update());                  // 声明序;组件自驱,引擎不解释
        } catch (Throwable t) {
            quarantine(t);                       // 单实例故障隔离 → end(EndResult.nowhere()) → 会话强制切回退模式
        }
    }

    /** MatchModeInstancePort#end 的落点:只置位与通知会话,不清场(ticker 下一 tick 处置,§3.1)。 */
    void requestEnd(EndResult result) {
        if (ending || closed) return;
        ending = true;
        session.instanceEnded(this, result);
    }

    /** ★ 幂等。切换阶段 B2 调用。 */
    @Override public void close() {
        if (closed) return;
        closed = true;
        components.inReverseDeclarationOrder().forEach(this::safeStop); // 逆序 stop;异常不中断(try/finally 链)
        components.reclaimRegisteredResources();                         // 组件登记的订阅/延时任务/显示统一回收
        bus.clear();
    }
    private void quarantine(Throwable t) { /* 记日志(原因 QUARANTINE)+ requestEnd(EndResult.nowhere()) */ }
}

            return;

        LOG.info("已加载模式: " + mode.id() + (mode.meta().recordsHistory() ? "" : "(不记史)"));
    }

    private void tickAll() {

                s.runPendingSwitch();             // ★ 切换只在此驱动(§3.5 防重入的机械保证)

    public static final class Builder<S> {
        public Builder<S> requires(Class<?> type);          // 缺 → seal 抛,点名双方
        public Builder<S> requiresOptional(Class<?> type);  // 缺 → 不报错,运行期 depOpt 自查
        public Builder<S> capabilities(String... tags);     // seal 成功注册,shutdown 摘除(§8.8 口径)
        public ServiceSpec<S> build();
    }
    String id(); Class<S> type(); Function<ServiceEnv, S> factory();
    List<Class<?>> required(); Set<String> capabilities();
}

// engine/service/ServiceRegistry.java —— 两阶段装配容器(★D2;核心零 Bukkit import)
public final class ServiceRegistry implements AutoCloseable {
    public ServiceRegistry(CapabilityRegistry capabilities);

    public record Context(SessionService sessions, ServiceRegistry services,

        reg.seal();                                                     //    失败 → SEVERE + setEnabled(false)

                                                                       //   ★v1.3-D5:此行废止——整段拓扑装配移宿主,框架 onEnable 零会话;
                                                                       //   宿主自调 SessionService.create(SessionSpec)(21-D5)
        ModeAssembler.discover();                                       // ⑤ 模式双发现

              .register(MiniGameApi.class, api, plugin, ServicePriority.Normal);

    }

// ═══════════════════════════════════════════════════════════════
// listener/StatsSignalListener.java —— ★v3.1 自 RoleSignalListener 拆出(该监听器属 SHDF 扩展线,已迁 Shdf 侧):伤害记账与角色无关
// ═══════════════════════════════════════════════════════════════
package com.hydrokarbs.listener;

public final class StatsSignalListener implements Listener {

    /** ignoreCancelled = true:宿主插件可能会取消伤害(免疫、格挡)。只记"真的扣血了"的。
     *  MONITOR:只记账不改状态,放最后拿最终伤害值。 */
    @EventHandler(priority = EventPriority.MONITOR, ignoreCancelled = true)
    public void onDamage(EntityDamageByEntityEvent event) {
        if (event.getEntity() instanceof Player victim) {
            sessions.sessionOf(victim.getUniqueId()).flatMap(Session::currentInstance)
                    .map(i -> i.component(StatsComponent.class))
                    .ifPresent(stats -> stats.addDamageTaken(victim.getUniqueId(), (long) event.getFinalDamage()));
        }
        if (event.getDamager() instanceof Player attacker) {
            sessions.sessionOf(attacker.getUniqueId()).flatMap(Session::currentInstance)
                    .map(i -> i.component(StatsComponent.class))
                    .ifPresent(stats -> stats.addDamageDealt(attacker.getUniqueId(), (long) event.getFinalDamage()));
        }
    }
}
